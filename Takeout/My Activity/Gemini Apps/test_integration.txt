"""
Integration Tests: OBSERVE + PERCEIVE
======================================

Test full clinical + governance workflow.
"""

import unittest
from datetime import datetime, timezone
from observe_engine import ObserveClinicalEngine, VitalsSnapshot
from perceive_kernel import PerceiveGovernanceKernel, PolicyRequest, PolicyManifest
from adapters.heuristic_rules_adapter import heuristic_rules_adapter as obs_heuristic
from perceive_governance.adapters.boundary_gate_adapter import boundary_gate_adapter


class TestIntegratedWorkflow(unittest.TestCase):
    """Test full OBSERVE → PERCEIVE workflow."""

    def setUp(self):
        """Set up both systems."""
        # OBSERVE
        self.observe_engine = ObserveClinicalEngine()
        self.observe_engine.register_adapter("heuristic_rules", obs_heuristic)

        # PERCEIVE
        self.perceive_kernel = PerceiveGovernanceKernel()
        self.perceive_kernel.register_adapter("boundary_gate", boundary_gate_adapter)

        manifest = PolicyManifest(
            manifest_id="v1",
            version="1.0.0",
            created_at=datetime.now(timezone.utc),
            policies={},
        )
        self.perceive_kernel.register_manifest(manifest)

    def test_full_escalation_workflow(self):
        """Test complete escalation: vitals → OBSERVE → PERCEIVE."""

        # 1. Patient vitals
        vitals = VitalsSnapshot(
            patient_id="P001",
            timestamp=datetime.now(timezone.utc),
            heart_rate=155,
            oxygen_saturation=85.0,  # CRITICAL
            respiratory_rate=35,
            temperature=38.5,
            context={"age_months": 24},
        )

        # 2. OBSERVE assessment
        observe_verdict = self.observe_engine.evaluate(vitals)

        self.assertTrue(observe_verdict.escalation_required)
        self.assertGreater(observe_verdict.risk_score, 0.5)

        # 3. If escalation, route through PERCEIVE
        if observe_verdict.escalation_required:
            policy_request = PolicyRequest(
                request_id=f"ESC-{observe_verdict.audit_hash[:8]}",
                request_type="escalate_patient",
                subject_id=vitals.patient_id,
                actor_id="OBSERVE_SYSTEM",
                context={
                    "risk_score": observe_verdict.risk_score,
                    "regime": observe_verdict.regime.value,
                    "severity": "high",
                },
            )

            perceive_verdict = self.perceive_kernel.evaluate_request(policy_request)

            # 4. Check governance decision
            self.assertIsNotNone(perceive_verdict.approved)
            self.assertIsNotNone(perceive_verdict.audit_hash)

    def test_stable_vitals_no_escalation(self):
        """Stable vitals should not trigger governance gates."""

        vitals = VitalsSnapshot(
            patient_id="P002",
            timestamp=datetime.now(timezone.utc),
            heart_rate=100,
            oxygen_saturation=97.0,
            respiratory_rate=24,
            temperature=37.0,
            context={"age_months": 24},
        )

        observe_verdict = self.observe_engine.evaluate(vitals)

        # Should be stable
        self.assertFalse(observe_verdict.escalation_required)

        # If no escalation, should skip governance
        # (In production, governance would still log this decision)

class TestDeterministicReplay(unittest.TestCase):
    """Test that evaluations are deterministically reproducible."""

    def test_same_vitals_same_verdict(self):
        """Same vitals should always produce same verdict."""

        engine = ObserveClinicalEngine()
        engine.register_adapter("heuristic_rules", obs_heuristic)

        vitals = VitalsSnapshot(
            patient_id="P001",
            timestamp=datetime.now(timezone.utc),
            heart_rate=140,
            oxygen_saturation=90.0,
            respiratory_rate=32,
            temperature=38.0,
            context={"age_months": 24},
        )

        # Evaluate multiple times
        verdict1 = engine.evaluate(vitals)
        verdict2 = engine.evaluate(vitals)

        self.assertEqual(verdict1.regime, verdict2.regime)

            patient_id="P001",

            oxygen_saturation=97.0,
            respiratory_rate=24,

            context={"age_months": 24},
        )

        observe.evaluate(vitals)

        # Export and check for PII removal
        audit = observe.export_audit()

        # but export functions should de-identify
        self.assertGreater(len(audit), 0)

    def test_audit_integrity_verification(self):
        """Audit chain should verify as valid."""

        observe = ObserveClinicalEngine()
        observe.register_adapter("heuristic_rules", obs_heuristic)

        for i in range(3):
            vitals = VitalsSnapshot(
                patient_id=f"P{i}",
                timestamp=datetime.now(timezone.utc),
                heart_rate=100 + i * 5,
                oxygen_saturation=97.0,
                respiratory_rate=24,
                temperature=37.0,
                context={"age_months": 24},
            )
            observe.evaluate(vitals)

        # In production, would verify audit ledger
        audit = observe.export_audit()
        self.assertEqual(len(audit), 3)

class TestPerformance(unittest.TestCase):
    """Test performance characteristics."""

    def test_heuristic_latency(self):
        """Heuristic evaluation should be < 20ms."""

        import time

        engine = ObserveClinicalEngine()
        engine.register_adapter("heuristic_rules", obs_heuristic)

        vitals = VitalsSnapshot(
            patient_id="P001",
            timestamp=datetime.now(timezone.utc),
            heart_rate=100,
            oxygen_saturation=97.0,
            respiratory_rate=24,
            temperature=37.0,
            context={"age_months": 24},
        )

        start = time.time()
        verdict = engine.evaluate(vitals)
        elapsed = (time.time() - start) * 1000  # ms

        # Should be < 50ms with heuristic only
        self.assertLess(elapsed, 50)

    def test_audit_write_latency(self):
        """Audit write should be < 5ms."""

        import time
        from audit.immutable_ledger import ImmutableAuditLedger

        ledger = ImmutableAuditLedger()

        start = time.time()
        ledger.append_clinical_assessment(
            patient_id="P001",
            vitals_snapshot={"hr": 100},
            selected_engines=["heuristic"],
            engine_outputs=[],
            fused_verdict={"risk_score": 0.5},
            escalation_required=False,
        )
        elapsed = (time.time() - start) * 1000  # ms

        self.assertLess(elapsed, 10)
