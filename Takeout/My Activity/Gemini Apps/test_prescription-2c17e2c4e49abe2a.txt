"""
test_prescription.py — Locks the prescription layer.

Groups:
  CONFIDENCE CLASSIFIER (unit) — the pure classify_confidence function, with
    hand-crafted fold/seed inputs, so every level boundary and the ghost path are
    deterministically locked.
  GATING — only clean wins become candidates; false wins and landmines are
    REJECTED_NOT_CLEAN; a clean win that fails replication is REJECTED_GHOST.
  HONESTY OF OUTPUT — confidence is always a band (never a lone number); nothing is
    client-ready without the Partner's REAL + CLEAN verdicts.
  FEEDBACK LOOP — projected-vs-actual scores HIT / PARTIAL / MISS correctly.
"""

from __future__ import annotations

import copy
import pytest

from reference_tree import build_reference_tree
from caller_model import PopulationGenerator, PopulationConfig
from ivr_tree import Node, NodeKind
from simulator import push_option_last, lead_with
from prescription_engine import (
    PrescriptionEngine, classify_confidence,
    MIN_MEANINGFUL_DELTA, LARGE_SUPPORT,
)
from prescription_types import (
    ConfidenceLevel, PrescriptionStatus, Prescription, PrescriptionOutcome,
    OutcomeVerdict, Verdict,
)


def _setup(n=12000, seed=11):
    pop = PopulationGenerator(PopulationConfig(n=n, seed=seed)).generate()
    observed = {c.caller_id: c.observed() for c in pop}
    return build_reference_tree(), pop, observed


def _trap_dispute(t):
    tt = t.clone()
    tt.nodes["dispute_deadend"] = Node("dispute_deadend", NodeKind.SELF_SERVE,
        prompt="filed, goodbye", resolve_prob=0.0, next_on_fail=None)
    tt.node("dispute_action").next_on_fail = "dispute_deadend"
    n = tt.node("dispute_menu")
    n.options = [o for o in n.options if o.target_id != "agent"]
    return tt


# ── CONFIDENCE CLASSIFIER (unit, deterministic) ───────────────────────────────

def test_classifier_no_effect_below_floor():
    """A full-population effect below the meaningful floor is NO_EFFECT."""
    level, repl, _, _ = classify_confidence(
        full_delta=0.001, fold_deltas=[0.001, 0.0, 0.002, -0.001, 0.001],
        seed_deltas=[0.001], support=12000)
    assert level == ConfidenceLevel.NO_EFFECT
    print("PASS: sub-floor effect classified NO_EFFECT")


def test_classifier_ghost_on_sign_flip():
    """Folds that flip sign => GHOST (does not replicate)."""
    level, repl, rationale, _ = classify_confidence(
        full_delta=0.012,
        fold_deltas=[0.03, -0.02, 0.04, -0.01, 0.02],  # mixed signs
        seed_deltas=[0.012, 0.010], support=12000)
    assert level == ConfidenceLevel.GHOST
    assert not repl
    print(f"PASS: sign-flipping folds classified GHOST")


def test_classifier_high_when_tight_and_supported():
    """Sign-consistent, tight spread, large support, stable seeds => HIGH."""
    level, repl, _, _ = classify_confidence(
        full_delta=0.020,
        fold_deltas=[0.019, 0.021, 0.020, 0.018, 0.022],  # tight
        seed_deltas=[0.020, 0.019, 0.021], support=LARGE_SUPPORT + 1000)
    assert level == ConfidenceLevel.HIGH
    assert repl
    print("PASS: tight, supported, stable effect classified HIGH")


def test_classifier_moderate_on_wider_spread():
    """Sign-consistent but a wider spread => MODERATE."""
    level, repl, _, _ = classify_confidence(
        full_delta=0.017,
        fold_deltas=[0.011, 0.029, 0.015, 0.020, 0.013],  # moderate spread
        seed_deltas=[0.017, 0.018, 0.020], support=12000)
    assert level == ConfidenceLevel.MODERATE
    assert repl
    print("PASS: wider-spread effect classified MODERATE")


def test_classifier_low_on_thin_support():
    """Sign-consistent, tight, but thin support => LOW."""
    level, repl, _, _ = classify_confidence(
        full_delta=0.020,
        fold_deltas=[0.019, 0.021, 0.020],
        seed_deltas=[0.020, 0.019], support=200)  # thin
    assert level in (ConfidenceLevel.LOW, ConfidenceLevel.MODERATE)
    # specifically, thin support must prevent HIGH
    assert level != ConfidenceLevel.HIGH
    print("PASS: thin support prevents a HIGH rating")


def test_classifier_always_returns_caveats():
    """Every assessment carries the 'this is a model estimate' caveat — the band is
    never presented as gospel."""
    _, _, _, caveats = classify_confidence(0.02, [0.02, 0.02, 0.02], [0.02], 12000)
    assert any("model estimate" in c for c in caveats)
    assert any("not a formal statistical" in c for c in caveats)
    print("PASS: confidence always carries honesty caveats")


# ── GATING ────────────────────────────────────────────────────────────────────

def test_clean_win_becomes_candidate():
    """A replicating clean win becomes a CANDIDATE with a confidence band."""
    tree, pop, observed = _setup()
    eng = PrescriptionEngine()
    out = eng.prescribe(tree, [("push agent last", push_option_last("entry", "agent"))],
                        pop, observed)
    p = out[0]
    assert p.status == PrescriptionStatus.CANDIDATE
    assert p.confidence is not None
    assert p.confidence.level in (ConfidenceLevel.LOW, ConfidenceLevel.MODERATE, ConfidenceLevel.HIGH)
    print(f"PASS: clean win -> CANDIDATE ({p.confidence.level.value})")


def test_false_win_rejected_not_clean():
    """A false win (containment up via fake containment) is REJECTED_NOT_CLEAN and
    never reaches confidence assessment."""
    tree, pop, observed = _setup()
    eng = PrescriptionEngine()
    out = eng.prescribe(tree, [("trap dispute", _trap_dispute)], pop, observed)
    p = out[0]
    assert p.status == PrescriptionStatus.REJECTED_NOT_CLEAN
    assert "FALSE WIN" in p.rejection_reason
    print("PASS: false win -> REJECTED_NOT_CLEAN")


def test_non_replicating_clean_win_rejected_ghost():
    """A change that is a clean win on the full population but fails replication is
    REJECTED_GHOST. (lead-with-balance is a tiny effect that does not replicate.)"""
    tree, pop, observed = _setup()
    eng = PrescriptionEngine()
    out = eng.prescribe(tree, [("lead with balance", lead_with("entry", "balance"))],
                        pop, observed)
    p = out[0]
    # it is either a ghost (doesn't replicate) or no-effect — both are rejected
    assert p.status in (PrescriptionStatus.REJECTED_GHOST,)
    assert p.confidence.level in (ConfidenceLevel.GHOST, ConfidenceLevel.NO_EFFECT)
    print(f"PASS: non-replicating clean win -> REJECTED_GHOST ({p.confidence.level.value})")


# ── HONESTY OF OUTPUT ─────────────────────────────────────────────────────────

def test_confidence_is_a_band_not_a_number():
    """A candidate's confidence must expose a band (fold spread), not just a point."""
    tree, pop, observed = _setup()
    eng = PrescriptionEngine()
    out = eng.prescribe(tree, [("push agent last", push_option_last("entry", "agent"))],
                        pop, observed)
    c = out[0].confidence
    lo, hi = c.band
    assert len(c.fold_deltas) >= 2
    assert lo <= c.full_delta <= hi or abs(hi - lo) >= 0   # band brackets/contains estimate
    assert "to" in c.describe()  # the rendered output shows a range
    print(f"PASS: confidence is a band ({lo*100:+.1f} to {hi*100:+.1f}pt), not a lone number")


def test_candidate_not_client_ready_without_partner():
    """A CANDIDATE is not client-ready until the Partner fills REAL + CLEAN."""
    tree, pop, observed = _setup()
    eng = PrescriptionEngine()
    out = eng.prescribe(tree, [("push agent last", push_option_last("entry", "agent"))],
                        pop, observed)
    p = out[0]
    assert p.status == PrescriptionStatus.CANDIDATE
    assert not p.client_ready, "candidate self-certified as client-ready (must not!)"
    # simulate the Partner signing off
    p.validation_verdict = Verdict.REAL
    p.fairness_verdict = Verdict.CLEAN
    assert p.client_ready, "should be client-ready once Partner validates REAL + CLEAN"
    print("PASS: candidate is client-ready only after Partner REAL + CLEAN")


def test_partner_seam_present():
    """Every prescription carries the two verdict slots the Partner fills."""
    tree, pop, observed = _setup()
    eng = PrescriptionEngine()
    out = eng.prescribe(tree, [("push agent last", push_option_last("entry", "agent")),
                               ("trap dispute", _trap_dispute)], pop, observed)
    for p in out:
        assert hasattr(p, "validation_verdict") and hasattr(p, "fairness_verdict")
    print("PASS: partner seam present on every prescription")


# ── FEEDBACK LOOP ─────────────────────────────────────────────────────────────

def test_feedback_hit():
    rec = PrescriptionOutcome("x", projected_delta=0.017, projected_band=(0.011, 0.029))
    rec.score(0.018)
    assert rec.verdict == OutcomeVerdict.HIT
    print("PASS: feedback HIT when actual lands in band")


def test_feedback_partial():
    rec = PrescriptionOutcome("x", projected_delta=0.017, projected_band=(0.011, 0.029))
    rec.score(0.045)
    assert rec.verdict == OutcomeVerdict.PARTIAL
    print("PASS: feedback PARTIAL when same direction, outside band")


def test_feedback_miss_wrong_direction():
    rec = PrescriptionOutcome("x", projected_delta=0.017, projected_band=(0.011, 0.029))
    rec.score(-0.01)
    assert rec.verdict == OutcomeVerdict.MISS
    print("PASS: feedback MISS when actual is wrong direction")


def test_feedback_miss_no_effect():
    rec = PrescriptionOutcome("x", projected_delta=0.017, projected_band=(0.011, 0.029))
    rec.score(0.001)
    assert rec.verdict == OutcomeVerdict.MISS
    print("PASS: feedback MISS when actual is negligible")
